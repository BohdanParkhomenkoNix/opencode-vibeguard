Create `src/known.js`:

```js
/** At most this many alternatives per regex: one alternation of thousands of needles falls off V8's fast path. */
export const CHUNK = 200

const escapeRegex = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")

/**
 * Applies fn to the text between placeholders only. A placeholder already in the text is a
 * finished replacement: a match inside it would wrap it in a second placeholder, and restore
 * would then give a tool the inner placeholder instead of the value.
 */
export function mapGaps(text, placeholders, fn) {
  let out = ""
  let last = 0
  for (const m of text.matchAll(placeholders)) {
    out += fn(text.slice(last, m.index)) + m[0]
    last = m.index + m[0].length
  }
  return out + fn(text.slice(last))
}

/**
 * Finds and replaces a set of literal strings, longest first, with one regex per CHUNK needles.
 * Later chunks only see the gaps between placeholders, so a shorter needle never splits a
 * replacement an earlier chunk made.
 */
export function knownMatcher(needles) {
  const sorted = [...new Set(needles)].filter(Boolean).sort((a, b) => b.length - a.length)
  const chunks = []
  for (let i = 0; i < sorted.length; i += CHUNK) {
    chunks.push(new RegExp(sorted.slice(i, i + CHUNK).map(escapeRegex).join("|"), "g"))
  }
  return {
    size: sorted.length,
    find(text) {
      for (const re of chunks) {
        re.lastIndex = 0
        const m = re.exec(text)
        if (m) return m[0]
      }
      return null
    },
    replaceIn(text, placeholders, fn) {
      let out = text
      for (const re of chunks) out = mapGaps(out, placeholders, (gap) => gap.replace(re, fn))
      return out
    },
  }
}
```

```js
  const originalOf = new Map()
```

```js
function leakIn(text, known, placeholderOf, patterns, prefix) {
  const original = known.find(text)
  if (original !== null) return { category: categoryOf(placeholderOf.get(original), prefix) }
  let category = null
  redactText(text, patterns, {
    getOrCreatePlaceholder(_original, ruleCategory) {
      category ??= ruleCategory
      return ""
    },
  })
  return category ? { category } : null
}

/**
 * Last line of defence before bytes leave the machine: anything the rules match, or any value
 * already known to be secret, in any string of the outgoing body, also with JSON escapes undone.
 * Reports the category only.
 */
export function findLeak(bodyText, patterns, session) {
  if (!bodyText) return null
  const placeholders = getPlaceholderRegex(session.prefix)
  const placeholderOf = new Map(session.knownEntries())
  const known = knownMatcher([...placeholderOf.keys()])
  for (const raw of stringsOf(bodyText)) {
    const text = raw.replace(placeholders, " ")
    for (const view of [text, ...escapedViews(text)]) {
      const leak = leakIn(view, known, placeholderOf, patterns, session.prefix)
      if (leak) return leak
    }
  }
  return null
}
```

```js
/**
 * Values the rules find in the views of a gap (JSON escapes undone), registered only when the gap
 * spells them in a form the replacement can find; a value seen only in a view would take a
 * max_mappings slot for nothing. Returns needle → placeholder.
 */
function learnFromViews(gap, patterns, session) {
  const learned = new Map()
  for (const view of escapedViews(gap)) {
    const candidates = []
    redactText(view, patterns, {
      getOrCreatePlaceholder(original, category) {
        candidates.push({ original, category })
        return ""
      },
    })
    for (const { original, category } of candidates) {
      const needles = [original, ...escapedForms(original)].filter((n) => gap.includes(n))
      if (needles.length === 0) continue
      const placeholder = session.getOrCreatePlaceholder(original, category)
      for (const n of needles) learned.set(n, placeholder)
    }
  }
  return learned
}

/**
 * Values learned from the views are replaced in every escaped form first, then the rules run on
 * what is left of the raw text. In the other order a rule whose class includes `/` masks only the
 * part of `abc\/def…` after the backslash, and `abc\/` reaches the provider.
 */
function rulePass(root, patterns, session) {
  const placeholders = getPlaceholderRegex(session.prefix)
  return mapStrings(root, (text) => mapGaps(text, placeholders, (gap) => {
    const learned = learnFromViews(gap, patterns, session)
    const masked = learned.size === 0 ? gap : knownMatcher([...learned.keys()]).replaceIn(gap, placeholders, (found) => learned.get(found))
    return mapGaps(masked, placeholders, (rest) => redactText(rest, patterns, session).text)
  }), SKIP_KEYS)
}
```

```js
const TEMP_SUFFIX = /^(\d+)\.[0-9a-f]{8}\.tmp$/

function isAlive(pid) {
  try {
    process.kill(pid, 0)
    return true
  } catch (error) {
    // EPERM: alive, owned by someone else.
    return error.code !== "ESRCH"
  }
}

/** A process killed between write and unlink leaves its temp file, holding key bytes, behind. */
function removeOrphanTemps(file, fs) {
  const dir = path.dirname(file)
  const prefix = `${path.basename(file)}.`
  let names
  try {
    names = fs.readdirSync(dir)
  } catch {
    return
  }
  for (const name of names) {
    if (!name.startsWith(prefix)) continue
    const m = name.slice(prefix.length).match(TEMP_SUFFIX)
    if (!m) continue
    const pid = Number(m[1])
    if (pid === process.pid || isAlive(pid)) continue
    try {
      fs.unlinkSync(path.join(dir, name))
    } catch {
      // Already gone.
    }
  }
}
```

```js
  const tmp = `${file}.${process.pid}.${randomBytes(4).toString("hex")}.tmp`
  let written = false
  try {
    try {
      fs.writeFileSync(tmp, randomBytes(KEY_BYTES), { mode: 0o600, flag: "wx" })
      written = true
    } catch (error) {
      throw new Error(`cannot create key file ${file}: ${codeOf(error)}`)
    }
    try {
      fs.linkSync(tmp, file)
    } catch (error) {
      // EEXIST: another process linked its key first, and that key wins.
      if (error.code !== "EEXIST") throw new Error(`cannot create key file ${file}: ${codeOf(error)}`)
    }
  } finally {
    // Only our own temp file: on a name collision the file belongs to someone else.
    if (written) {
      try {
        fs.unlinkSync(tmp)
      } catch {
        // Already gone.
      }
    }
  }
}

/** The key turns placeholders back into guesses for short secrets: nobody else may read it. */
function ensureOwnerOnly(file, fs) {
  let mode
  try {
    mode = fs.statSync(file).mode
  } catch (error) {
    throw new Error(`cannot stat key file ${file}: ${codeOf(error)}`)
  }
  if ((mode & 0o077) === 0) return
  try {
    fs.chmodSync(file, 0o600)
  } catch (error) {
    throw new Error(`key file ${file} is accessible by others and chmod failed: ${codeOf(error)}`)
  }
}
```

```js
/**
 * An exact entry, or an address/host under that domain: `example.com` covers `a@example.com`,
 * `a@mail.example.com` and `api.example.com`. A credential such as `user:pass@localhost` is not
 * address-shaped, so the domain rule leaves it masked. Entries come trimmed and lowercased from
 * buildPatternSet.
 */
function isExcluded(original, exclude) {
  if (exclude.size === 0) return false
  const value = original.toLowerCase()
  for (const x of exclude) {
    if (value === x) return true
    if (value.endsWith(`@${x}`) && EMAIL_LOCAL_PART.test(value.slice(0, -(x.length + 1)))) return true
    if (value.endsWith(`.${x}`) && HOST_LABELS.test(value.slice(0, -(x.length + 1)))) return true
```
