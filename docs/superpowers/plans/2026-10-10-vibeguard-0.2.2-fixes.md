`src/key.js` целиком:

```js
import { randomBytes } from "node:crypto"
import * as nodeFs from "node:fs"
import os from "node:os"
import path from "node:path"

export const KEY_BYTES = 32

export function defaultKeyFile() {
  return path.join(os.homedir(), ".config", "opencode", "vibeguard.key")
}

/**
 * Written aside and linked into place, so no process ever reads a half-written key.
 * If another process links its key first, that key wins.
 */
function createKey(file, fs) {
  fs.mkdirSync(path.dirname(file), { recursive: true, mode: 0o700 })
  const tmp = `${file}.${process.pid}.${randomBytes(4).toString("hex")}.tmp`
  try {
    fs.writeFileSync(tmp, randomBytes(KEY_BYTES), { mode: 0o600, flag: "wx" })
    fs.linkSync(tmp, file)
  } catch (error) {
    if (error.code !== "EEXIST") throw new Error(`cannot create key file ${file}: ${error.code}`)
  } finally {
    try {
      fs.unlinkSync(tmp)
    } catch {
      // Nothing to remove: the temp file was never written.
    }
  }
}

/**
 * HMAC key shared by every session and every OpenCode restart, so a secret keeps its
 * placeholder across them. Created on first use with owner-only permissions.
 * `fs` is replaceable for tests only.
 */
export function loadKey(file = defaultKeyFile(), fs = nodeFs) {
  let key
  try {
    key = fs.readFileSync(file)
  } catch (error) {
    if (error.code !== "ENOENT") throw new Error(`cannot read key file ${file}: ${error.code}`)
    createKey(file, fs)
    key = fs.readFileSync(file)
  }
  if (key.length !== KEY_BYTES) throw new Error(`key file ${file} must hold ${KEY_BYTES} bytes, found ${key.length}`)
  return key
}
```

const MAX_JSON_DEPTH = 3

function parseNested(text) {
  const head = text.trimStart()[0]
  if (head !== "{" && head !== "[") return undefined
  try {
    return JSON.parse(text)
  } catch {
    return undefined
  }
}

function collect(value, out, depth) {
  mapStrings(value, (text) => {
    out.push(text)
    if (depth < MAX_JSON_DEPTH) {
      const nested = parseNested(text)
      if (nested !== undefined) collect(nested, out, depth + 1)
    }
    return text
  })
}

function stringsOf(bodyText) {
  let parsed
  try {
    parsed = JSON.parse(bodyText)
  } catch {
    return [bodyText]
  }
  if (typeof parsed === "string") return [parsed]
  const out = []
  // Decoded strings, not the raw body: a secret with a quote or a newline is escaped in JSON
  // and would slip past a substring search of the wire text. Strings that are JSON themselves
  // (tool_calls[].function.arguments) are decoded again for the same reason.
  collect(parsed, out, 0)
  return out
}

/** An exact entry, or a match under that domain: `example.com` covers `a@example.com` and `a@mail.example.com`. */
function isExcluded(original, exclude) {
  if (exclude.size === 0) return false
  const value = original.toLowerCase()
  for (const entry of exclude) {
    const x = entry.toLowerCase()
    if (!x) continue
    if (value === x || value.endsWith(`@${x}`) || value.endsWith(`.${x}`)) return true
  }
  return false
}
