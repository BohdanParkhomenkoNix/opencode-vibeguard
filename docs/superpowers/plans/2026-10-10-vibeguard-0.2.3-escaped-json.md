`src/escapes.js`:

```js
/** How many layers of JSON string escaping are undone (views) or applied (forms). */
export const MAX_ESCAPE_DEPTH = 3

const ESCAPE = /\\(?:u([0-9a-fA-F]{4})|(["\\/bfnrt]))/g
const SIMPLE = { '"': '"', "\\": "\\", "/": "/", b: "\b", f: "\f", n: "\n", r: "\r", t: "\t" }

function unescapeOnce(text) {
  return text.replace(ESCAPE, (_, hex, ch) => (hex ? String.fromCharCode(parseInt(hex, 16)) : SIMPLE[ch]))
}

/**
 * The text with JSON string escapes undone, one entry per level that changes it. Tool output
 * such as `read` of a notebook is escaped JSON wrapped in line numbers, not JSON itself, so a
 * context rule (`password = "…"`) only matches once `\"` is undone.
 */
export function escapedViews(text) {
  if (!text.includes("\\")) return []
  const views = []
  let current = text
  for (let i = 0; i < MAX_ESCAPE_DEPTH && current.includes("\\"); i++) {
    const next = unescapeOnce(current)
    if (next === current) break
    views.push(next)
    current = next
  }
  return views
}

const jsonEscape = (s) => JSON.stringify(s).slice(1, -1)
const slashEscape = (s) => s.replaceAll("/", "\\/")
const asciiEscape = (s) => s.replace(/[\u0080-\uffff]/g, (c) => `\\u${c.charCodeAt(0).toString(16).padStart(4, "0")}`)

/**
 * How a value can be spelled inside escaped text: JSON-escaped once to three times, each level
 * also with `\/` (PHP) and `\uXXXX` for non-ASCII (Python ensure_ascii, Jupyter).
 */
export function escapedForms(value) {
  const forms = new Set()
  let level = value
  for (let i = 0; i < MAX_ESCAPE_DEPTH; i++) {
    level = jsonEscape(level)
    forms.add(level).add(slashEscape(level)).add(asciiEscape(level))
  }
  forms.delete(value)
  return [...forms]
}
```

```js
function leakIn(text, known, patterns, prefix) {
  for (const [original, placeholder] of known) {
    if (text.includes(original)) return { category: categoryOf(placeholder, prefix) }
  }
  let category = null
  redactText(text, patterns, {
    getOrCreatePlaceholder(_original, ruleCategory) {
      category ??= ruleCategory
      return ""
    },
  })
  return category ? { category } : null
}

/**
 * Last line of defence before bytes leave the machine: anything the rules match, or any value
 * already known to be secret, in any string of the outgoing body, also with JSON escapes undone.
 * Reports the category only.
 */
export function findLeak(bodyText, patterns, session) {
  if (!bodyText) return null
  const placeholders = getPlaceholderRegex(session.prefix)
  const known = [...session.knownEntries()]
  for (const raw of stringsOf(bodyText)) {
    const text = raw.replace(placeholders, " ")
    for (const view of [text, ...escapedViews(text)]) {
      const leak = leakIn(view, known, patterns, session.prefix)
      if (leak) return leak
    }
  }
  return null
}
```

```js
/**
 * Rules run on the text and, to learn values only, on its views with JSON escapes undone:
 * `password = \"…\"` in a notebook read with `read` only matches once `\"` is undone. The
 * known-value pass then replaces the value in whatever escaped form it takes in the text.
 */
function rulePass(root, patterns, session) {
  const placeholders = getPlaceholderRegex(session.prefix)
  return mapStrings(root, (text) => mapGaps(text, placeholders, (gap) => {
    for (const view of escapedViews(gap)) redactText(view, patterns, session)
    return redactText(gap, patterns, session).text
  }), SKIP_KEYS)
}

/**
 * Replaces every value the session already knows, and every escaped form of it, with its
 * existing placeholder, longest first. A context rule (a lookbehind) learns a value from one
 * occurrence, but the same value turns up bare or escaped in tool output, and the guard blocks
 * any known value it finds.
 */
function knownPass(root, session) {
  const originalOf = new Map()
  for (const [original] of session.knownEntries()) {
    if (!original) continue
    originalOf.set(original, original)
    for (const form of escapedForms(original)) if (!originalOf.has(form)) originalOf.set(form, original)
  }
  if (originalOf.size === 0) return 0
  const needles = [...originalOf.keys()].sort((a, b) => b.length - a.length)
  const known = new RegExp(needles.map((o) => o.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")).join("|"), "g")
  const placeholders = getPlaceholderRegex(session.prefix)
  return mapStrings(root, (text) => mapGaps(text, placeholders, (gap) => gap.replace(known, (found) => session.lookupReverse(originalOf.get(found)))), SKIP_KEYS)
}
```
